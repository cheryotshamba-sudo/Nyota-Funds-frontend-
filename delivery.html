// ========================================== function updateFee(){ const selected = document.querySelector( 'input[name="delivery"]:checked' ); if(selected){ document.getElementById("deliveryFee").innerText = selected.value; } } // ========================================== // PAY DELIVERY // ========================================== async function payDelivery(){ const selected = document.querySelector( 'input[name="delivery"]:checked' ); if(!selected){ alert("Please select delivery method."); return; } const phone = localStorage.getItem("phoneNumber"); if(!phone){ alert("Phone number not found."); return; } const amount = Number(selected.value); const status = document.getElementById("status"); const button = document.getElementById("payButton"); button.disabled = true; button.innerText = "Processing..."; status.innerText = "Sending STK Push..."; try{ const response = await fetch( "https://nyota-funds-backend.onrender.com/stk-push", { method:"POST", headers:{ "Content-Type":"application/json" }, body:JSON.stringify({ phone:phone, amount:amount }) } ); const data = await response.json(); console.log( "STK RESPONSE:", data ); if(!response.ok){ throw new Error( data.message || "STK Push failed" ); } // ================================== // GET PAYLOR TRANSACTION ID // ================================== const transactionId = data.transactionId || data.checkout_request_id || data.data?.transactionId || data.data?.checkout_request_id; if(!transactionId){ console.error( "No transaction ID found:", data ); throw new Error( "Payment reference was not returned." ); } console.log( "PAYMENT ID:", transactionId ); status.innerText = "📱 STK Push sent. Please enter your M-PESA PIN."; // Start checking payment checkPayment(transactionId); }catch(error){ console.error( "STK ERROR:", error ); button.disabled = false; button.innerText = "Pay Delivery Charges"; status.innerText = "❌ Failed to send STK Push."; } } // ========================================== // CHECK PAYMENT STATUS // ========================================== function checkPayment(transactionId){ console.log( "Checking payment:", transactionId ); const interval = setInterval(async()=>{ try{ // ================================== // DEBUG TRANSACTION ID // ================================== console.log( "SENDING PAYMENT STATUS:", { transactionId: transactionId } ); const response = await fetch( "https://nyota-funds-backend.onrender.com/payment-status", { method:"POST", headers:{ "Content-Type":"application/json" }, body:JSON.stringify({ checkout_request_id: transactionId, transactionId: transactionId, transaction_id: transactionId }) } ); const result = await response.json(); console.log( "PAYMENT STATUS:", result ); // ================================== // GET PAYMENT STATUS // ================================== const paymentStatus = String( result.status || result.data?.status || result.data?.data?.status || "" ).toLowerCase(); // ================================== // COMPLETED // ================================== if( paymentStatus === "completed" || paymentStatus === "success" ){ clearInterval(interval); document.getElementById("status").innerText = "✅ Delivery charges paid successfully."; setTimeout(()=>{ window.location.href = "commitment.html"; },1000); } // ================================== // CANCELLED // ================================== else if( paymentStatus === "cancelled" ){ clearInterval(interval); document.getElementById("status").innerText = "❌ Payment cancelled."; document.getElementById("payButton").disabled = false; document.getElementById("payButton").innerText = "Pay Delivery Charges"; } // ================================== // FAILED // ================================== else if( paymentStatus === "failed" ){ clearInterval(interval); document.getElementById("status").innerText = "❌ Payment failed."; document.getElementById("payButton").disabled = false; document.getElementById("payButton").innerText = "Pay Delivery Charges"; } // ================================== // PENDING // ================================== else{ document.getElementById("status").innerText = "⏳ Payment received. Confirming payment..."; } }catch(error){ console.error( "STATUS ERROR:", error ); } },3000); }
